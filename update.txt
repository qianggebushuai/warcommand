第三步：游戏核心逻辑重构 (Logic & Win)
1. 修改 units.py：索敌逻辑 (支持混战)
AI 和自动攻击不能只认 "player" 或 "enemy"。

Python

# units.py -> Unit.find_nearest_enemy
def find_nearest_enemy(self):
    nearest = None
    min_dist = self.attack_range * 1.5 if self.attack_range > 0 else 300
    
    # [核心修改] 遍历所有单位，只要队伍不一样且不是中立，就是敌人
    targets = self.game.all_units + self.game.all_buildings
    
    for t in targets:
        if not t.alive: continue
        if t.team == self.team: continue # 队友
        if t.team == "neutral": 
            # 特殊处理：只攻击特定的中立建筑(如摩天大楼)，不打油井
            if getattr(t, 'b_type', '') != "skyscraper": continue
            
        # 距离判断
        dist = self.pos.distance_to(t.pos)
        if dist < min_dist:
            # 空军判定
            if t.is_air and not self.can_attack_air: continue
            
            min_dist = dist
            nearest = t
            
    return nearest
