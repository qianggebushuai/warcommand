class Mi24(Unit):
    def __init__(self, game, pos, team):
        super().__init__(game, pos, team)
        # ... (属性) ...
        self.rotor_angle = 0
        
        # [加载图片]
        # 1. 机身
        path_body = resource_path("picture/unit_mi24_body.png")
        if os.path.exists(path_body):
            raw_body = pygame.image.load(path_body).convert_alpha()
            self.body_img = pygame.transform.scale(raw_body, (80, 40)) # 调整大小
        else:
            self.body_img = self.generate_fallback_body() # 没图时的备选
            
        # 2. 旋翼
        path_rotor = resource_path("picture/unit_mi24_rotor.png")
        if os.path.exists(path_rotor):
            raw_rotor = pygame.image.load(path_rotor).convert_alpha()
            self.rotor_img = pygame.transform.scale(raw_rotor, (100, 100))
        else:
            self.rotor_img = self.generate_fallback_rotor()

    def update(self):
        super().update()
        # 旋转逻辑
        self.rotor_angle = (self.rotor_angle + 40) % 360 # 转得快点

    def draw(self, surface, offset, zoom):
        screen_pos = to_screen(self.pos, offset, zoom)
        
        # 1. 画机身 (随 self.angle 转)
        s_body = scale_img(self.body_img, zoom)
        
        # 注意: 这里的 angle 旋转方向
        rot_body = pygame.transform.rotate(s_body, self.angle) 
        body_rect = rot_body.get_rect(center=screen_pos)
        surface.blit(rot_body, body_rect)
        
        # 2. 画旋翼 (随 self.rotor_angle 转)
        # 旋翼中心可能需要相对于机身中心偏移 (比如 Mi24 的旋翼轴偏前)
        # 假设旋翼轴在机身中心前方 10 像素
        rotor_offset = vec(10 * zoom, 0).rotate(self.angle) # 随机身转
        rotor_pos = (screen_pos[0] + rotor_offset.x, screen_pos[1] + rotor_offset.y)
        
        s_rotor = scale_img(self.rotor_img, zoom)
        rot_rotor = pygame.transform.rotate(s_rotor, self.rotor_angle)
        rotor_rect = rot_rotor.get_rect(center=rotor_pos)
        surface.blit(rot_rotor, rotor_rect)
        
        # 3. 尾桨 (可选，Mi24 有个小尾桨)
        # 画个小的旋转圆圈在尾部
        
        self.draw_health_and_marker(surface, offset, zoom)

    def generate_fallback_body(self):
        # 原来的绘制代码放这里
        s = pygame.Surface((80, 40), pygame.SRCALPHA)
        pygame.draw.rect(s, (50, 60, 50), (0, 10, 80, 20))
        return s

    def generate_fallback_rotor(self):
        s = pygame.Surface((100, 100), pygame.SRCALPHA)
        pygame.draw.line(s, (20,20,20), (50,0), (50,100), 2)
        pygame.draw.line(s, (20,20,20), (0,50), (100,50), 2)
        return s
