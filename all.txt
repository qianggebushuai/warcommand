    def update_enemy_ai(self):
        if not self.enemy_base.alive: return
        if self.enemy_exp < 10000: self.enemy_exp += self.ai_income_rate 
        self.enemy_build_timer += 1
        if self.enemy_build_timer > AI_BUILD_INTERVAL:
            self.enemy_build_timer = 0
            self.ai_attempt_construction()
        self.ai_produce_units()
        self.ai_tactical_command()
        self.ai_use_support()
        self.ai_capture_neutral()

    def ai_capture_neutral(self):
        neutrals = [b for b in self.all_buildings if isinstance(b, (OilDerrick, MercenaryFactory)) and b.team != "enemy"]
        if not neutrals: return
        infantry = [u for u in self.all_units if u.team == "enemy" and u.ai_mode == "GUARD" and not u.is_air]
        if infantry:
            # (4) 增加 AI 探索范围，不仅仅是随机选，而是选最近的
            # 这里简化为随机选一个去占领
            target = random.choice(neutrals)
            unit = random.choice(infantry)
            unit.target_pos = target.pos

    def ai_attempt_construction(self):
        buildings = [b.b_type for b in self.all_buildings if hasattr(b, 'b_type') and b.team == "enemy"]
        build_order = []
        if "mining" not in buildings: build_order.append(("mining", MINING_COST))
        if "factory" not in buildings: build_order.append(("factory", 0)) 
        if "radar" not in buildings: build_order.append(("radar", RADAR_COST))
        if "airfield" not in buildings: build_order.append(("airfield", AIRFIELD_COST))
        if "research" not in buildings: build_order.append(("research", RESEARCH_COST))
        if "aa_turret" not in buildings: build_order.append(("aa_turret", AA_TURRET_COST))
        if "missile_def" not in buildings: build_order.append(("missile_def", MISSILE_DEFENSE_COST))
        if "laser_turret" not in buildings: build_order.append(("laser_turret", LASER_TURRET_COST))
        if not build_order:
            if random.random() < 0.5: build_order.append(("aa_turret", AA_TURRET_COST))
            else: build_order.append(("mining", MINING_COST))
        
        bases = [b for b in self.all_buildings if hasattr(b, 'b_type') and b.team == "enemy" and b.b_type == "base" and b.alive]
        if not bases: return

        for b_type, cost in build_order:
            if self.enemy_exp >= cost:
                construction_base = random.choice(bases)
                offset = vec(random.randint(200, 600), 0).rotate(random.randint(0, 360))
                build_pos = construction_base.pos + offset
                build_pos.x = max(100, min(build_pos.x, WORLD_WIDTH-100))
                build_pos.y = max(100, min(build_pos.y, WORLD_HEIGHT-100))
                self.enemy_exp -= cost
                eng = Engineer(self, construction_base.pos + vec(0, 50), "enemy", build_pos)
                eng.target_type = b_type
                self.all_units.append(eng)
                break 

    def ai_produce_units(self):
        my_buildings = [b for b in self.all_buildings if hasattr(b, 'b_type') and b.team == "enemy"]
        enemy_count = len([u for u in self.all_units if u.team == "enemy"])
        if enemy_count >= 80: return 
        game_time = self.frame_count / 60

        for b in my_buildings:
            if b.b_type == "factory":
                r = random.random()
                unit, cost = None, 0
                tank_prob = 0.3 if game_time < 300 else 0.7
                if r < 0.3 and self.enemy_exp >= RIFLEMAN_COST: unit, cost = "rifleman", RIFLEMAN_COST
                elif r < 1.0 - tank_prob and self.enemy_exp >= ROCKET_COST: unit, cost = "rocket", ROCKET_COST
                elif self.enemy_exp >= ARMORED_COST: unit, cost = "armored", ARMORED_COST
                # AI 也会造 M1A2 (低概率)
                if game_time > 600 and random.random() < 0.1 and self.enemy_exp >= M1A2_COST:
                    unit, cost = "m1a2", M1A2_COST

                if unit and random.randint(0, 100) < 2:
                    self.enemy_exp -= cost
                    self.spawn_unit(unit, b.pos + vec(0, 50), "enemy")

            elif b.b_type == "airfield":
                if not getattr(b, 'is_upgraded', False) and self.enemy_exp >= UPGRADE_AIRFIELD_COST:
                    self.enemy_exp -= UPGRADE_AIRFIELD_COST
                    b.is_upgraded = True
                r = random.random()
                unit, cost = None, 0
                kirov_chance = 0.3 if game_time > 300 else 0.05
                if getattr(b, 'is_upgraded', False) and r < kirov_chance and self.enemy_exp >= KIROV_COST: unit, cost = "kirov", KIROV_COST
                elif r < 0.6 and self.enemy_exp >= HELICOPTER_COST: unit, cost = "helicopter", HELICOPTER_COST
                elif self.enemy_exp >= WZ10_COST: unit, cost = "wz10", WZ10_COST
                if unit and random.randint(0, 150) < 2:
                    self.enemy_exp -= cost
                    self.spawn_unit(unit, b.pos + vec(0, 60), "enemy")

    def ai_tactical_command(self):
        combat_units = [u for u in self.all_units if u.team == "enemy" and not isinstance(u, (Engineer, Harvester, CruiseMissile))]
        if not combat_units: return
        
        bases = [b for b in self.all_buildings if hasattr(b, 'b_type') and b.team == "enemy" and b.b_type == "base" and b.alive]
        if not bases: return

        squads = {b: [] for b in bases}
        for u in combat_units:
            nearest_base = min(bases, key=lambda b: u.pos.distance_squared_to(b.pos))
            squads[nearest_base].append(u)

        for base, squad in squads.items():
            if not squad: continue
            if not hasattr(base, 'lane_state'): base.lane_state = "GATHER"
            if base.lane_state == "GATHER":
                if len(squad) >= AI_WAVE_SIZE:
                    base.lane_state = "CHARGE"
                    self.particles.add(FloatingText(base.pos, "ASSAULT!", RED, 120))
            elif base.lane_state == "CHARGE":
                if len(squad) < 2: base.lane_state = "GATHER"
            vec_to_player = (self.player_base.pos - base.pos).normalize()
            rally_point = base.pos + vec_to_player * 300
            for u in squad:
                if base.lane_state == "GATHER":
                    u.ai_mode = "GUARD"
                    if u.pos.distance_to(rally_point) > 200: u.target_pos = rally_point
                else:
                    u.ai_mode = "ATTACK"
                    if not u.target_entity and not u.target_pos: u.target_pos = self.player_base.pos

    def ai_use_support(self):
        has_radar = any(hasattr(b, 'b_type') and b.b_type == "radar" and b.team == "enemy" for b in self.all_buildings)
        if has_radar and self.enemy_exp > 2000 and random.randint(0, 600) == 0:
            self.enemy_exp -= MISSILE_COST
            target = self.player_base.pos
            offset = vec(random.randint(-100, 100), random.randint(-100, 100))
            self.all_units.append(CruiseMissile(self, target + offset, "enemy"))
            self.show_alert("WARNING: ENEMY MISSILE DETECTED!")
        if has_radar and self.enemy_exp > 1500 and random.randint(0, 800) == 0:
            self.enemy_exp -= AC130_COST
            self.all_units.append(AC130(self, self.player_base.pos, "enemy"))
            self.show_alert("WARNING: ENEMY AC-130 ABOVE!")

    def issue_command(self, world_pos):
        target_entity = None
        for b in self.all_buildings:
            if b.team != "player" and b.rect.collidepoint(world_pos):
                target_entity = b; break
        if not target_entity:
            for u in self.all_units:
                if u.team != "player" and u.rect.collidepoint(world_pos):
                    target_entity = u; break
        if target_entity:
            if target_entity.is_air and not self.selected_entity.can_attack_air:
                self.selected_entity.move_to(target_entity.pos)
            else: self.selected_entity.set_target(target_entity)
        else: self.selected_entity.move_to(world_pos)

    def update(self):
        if self.state == "PLAYING":
            if self.paused: return
            self.frame_count += 1 
            
            # (5) 轰炸模式逻辑 (修改为制导)
            if self.bombardment_mode and self.frame_count % 120 == 0: 
                # 1. 随机生成目标坐标 (全图范围)
                # 留出一点边距，避免炸到地图最边缘看不到
                target_x = random.randint(100, WORLD_WIDTH - 100)
                target_y = random.randint(100, WORLD_HEIGHT - 100)
                target_pos = vec(target_x, target_y)

                # 2. 创建巡航导弹 (中立)
                missile = CruiseMissile(self, target_pos, "neutral")
                
                # 3. 设置导弹出生点
                # 从目标点上方较远的地方生成，带一点随机水平偏移，看起来像从四面八方飞来
                spawn_offset = vec(random.randint(-1000, 1000), -1500)
                missile.pos = target_pos + spawn_offset
                
                self.all_units.append(missile)
                
                if self.frame_count % 600 == 0: 
                    self.show_alert("MISSILE SHOWER INCOMING!")
            if self.game_over: 
                self.state = "GAMEOVER"
                return
            if self.cmd_exp < self.max_cmd_exp: self.cmd_exp += CMD_EXP_RATE
            
            if not self.player_base.alive: self.game_over = True; self.winner = "ENEMY"
            else:
                enemy_alive = any(b.team == "enemy" and b.b_type == "base" and b.alive for b in self.all_buildings if hasattr(b, 'b_type'))
                if not enemy_alive: self.game_over = True; self.winner = "PLAYER"
            if self.game_over and not self.data_saved:
                is_win = (self.winner == "PLAYER")
                # 保存数据到数据库
                self.db.update_stats(self.current_user, is_win, int(self.cmd_exp))
                print("Data Saved to MySQL!")
                self.data_saved = True
            # 维护 AI 主基地引用
            if not self.enemy_base.alive:
                for b in self.all_buildings:
                    if hasattr(b, 'b_type') and b.team == "enemy" and b.b_type == "base" and b.alive:
                        self.enemy_base = b
                        break

            self.update_enemy_ai()
            for u in self.all_units[:]:
                u.update()
                if not u.alive: self.all_units.remove(u)
            for b in self.all_buildings[:]:
                b.update()
                if not b.alive: self.all_buildings.remove(b)
            for p in self.projectiles[:]:
                p.update()
                if not p.alive: self.projectiles.remove(p)
            self.particles.update()
            self.floating_texts.update()
            if self.active_alert:
                self.active_alert[1] -= 1
                if self.active_alert[1] <= 0: self.active_alert = None

    def draw_world(self):
        self.screen.fill(GRASS_COLOR)
        # 1. 背景
        view_rect = pygame.Rect(self.camera_offset.x, self.camera_offset.y, SCREEN_WIDTH/self.zoom, SCREEN_HEIGHT/self.zoom)
        safe_x = max(0, int(view_rect.x))
        safe_y = max(0, int(view_rect.y))
        safe_w = min(WORLD_WIDTH - safe_x, int(view_rect.width) + 1)
        safe_h = min(WORLD_HEIGHT - safe_y, int(view_rect.height) + 1)
        if safe_w > 0 and safe_h > 0:
            sub_bg = self.bg_surface.subsurface((safe_x, safe_y, safe_w, safe_h))
            scaled_bg = pygame.transform.scale(sub_bg, (int(safe_w * self.zoom), int(safe_h * self.zoom)))
            screen_x = int((safe_x - self.camera_offset.x) * self.zoom)
            screen_y = int((safe_y - self.camera_offset.y) * self.zoom)
            self.screen.blit(scaled_bg, (screen_x, screen_y))

        # 2. 实体
        for b in self.all_buildings: b.draw(self.screen, self.camera_offset, self.zoom)
        for u in self.all_units:
            if u.is_air:
                spos = to_screen(u.pos, self.camera_offset, self.zoom)
                sw = scale(20, self.zoom)
                sh = scale(10, self.zoom)
                offset_y = 50 if isinstance(u, (AC130, KirovAirship, CruiseMissile)) else 30
                pygame.draw.ellipse(self.screen, (0,0,0,100), (spos[0]-sw//2, spos[1]+scale(offset_y, self.zoom), sw, sh))
        for u in self.all_units: u.draw(self.screen, self.camera_offset, self.zoom)
        for p in self.projectiles: p.draw(self.screen, self.camera_offset, self.zoom)
        for p in self.particles: p.draw(self.screen, self.camera_offset, self.zoom)
        for t in self.floating_texts: t.draw(self.screen, self.camera_offset, self.zoom)

        # 3. 选中与菜单
        if self.selected_entity and self.selected_entity.alive:
            spos = to_screen(self.selected_entity.pos, self.camera_offset, self.zoom)
            srad = scale(self.selected_entity.radius + 10, self.zoom)
            pygame.draw.circle(self.screen, YELLOW, spos, srad, 2)
            if self.selected_entity.team == "player" and isinstance(self.selected_entity, Building):
                self.draw_building_menu(self.selected_entity)

        # 4. 建造预览
        if self.place_building_type:
            m_screen_pos = pygame.mouse.get_pos()
            if self.player_base:
                base_screen_pos = to_screen(self.player_base.pos, self.camera_offset, self.zoom)

            # 简单的预览文字
            txt = self.font.render(f"Build: {self.place_building_type}", True, WHITE)
            self.screen.blit(txt, (m_screen_pos[0]+20, m_screen_pos[1]+20))

        # 5. 瞄准光标
        if self.call_airstrike_mode:
            m_pos = pygame.mouse.get_pos()
            pygame.draw.circle(self.screen, RED, m_pos, int(AC130_ORBIT_RADIUS * self.zoom), 1)
            t = self.font.render("AC-130 Target", True, RED)

    def handle_menu_click(self, screen_m_pos, screen_center_pos):
        c = vec(screen_center_pos)
        def check(off_x, off_y, cost, action_cb):
            btn_r = pygame.Rect(c.x + off_x - 20, c.y + off_y - 20, 40, 40)
            if btn_r.collidepoint(screen_m_pos):
                if cost > 0 and self.cmd_exp < cost: return True
                if cost > 0: self.cmd_exp -= cost
                action_cb()
                return True
            return False
        def check_no_deduct(off_x, off_y, cost, mode_setter):
            btn_r = pygame.Rect(c.x + off_x - 20, c.y + off_y - 20, 40, 40)
            if btn_r.collidepoint(screen_m_pos):
                if self.cmd_exp >= cost: mode_setter()
                return True
            return False
        world_center = self.selected_entity.pos 
        world_center = self.selected_entity.pos 
        if self.selected_entity.b_type == "base":
            if self.base_menu_mode == "main":
                if check(0, -80, 0, lambda: setattr(self, 'base_menu_mode', 'recruit')): return True
                
                # ▼▼▼ 确保这里传入了 COST 常量 ▼▼▼
                if check(80, -20, 0, lambda: setattr(self, 'place_building_type', 'factory')): return True
                # 这里有个逻辑问题：点击“建造工厂”按钮时不应该扣费，
                # 而是在 spawn_engineer 时扣费。
                # 但目前的 spawn_engineer 没有检查费用的逻辑，它是直接生成的。
                # 最好的做法是在这里扣费，如果钱不够就不进入建造模式。
                
                # --- 修正逻辑：点击建筑按钮时扣费 ---
                # 注意：这意味着如果你点了又不造（右键取消），钱就白扣了。
                # 更好的逻辑是：进入建造模式不扣费，点击地图放置时扣费。
                
                # 既然你想要“添加费用消耗”，我们采用“点击地图放置时扣费”的逻辑
                # 所以这里的 check 费用设为 0 是对的 (只是切换模式)
                
                # 等等，之前的代码是在 spawn_engineer 之前没有检查费用的。
                # 我们需要在 handle_input 的 MOUSEBUTTONDOWN 里，点击空地时检查费用。
                
                # 所以，这里保持 check(..., 0, ...) 是为了切换模式。
                # 我们去 handle_input 里修改。
                
                if check(80, -20, 0, lambda: setattr(self, 'place_building_type', 'factory')): return True
                if check(0, 80, 0, lambda: setattr(self, 'place_building_type', 'radar')): return True
                if check(-60, 60, 0, lambda: setattr(self, 'place_building_type', 'mining')): return True
                if check(60, 60, 0, lambda: setattr(self, 'place_building_type', 'airfield')): return True
                if check(-80, -20, 0, lambda: setattr(self, 'place_building_type', 'research')): return True
            elif self.base_menu_mode == "recruit":
                if check(-60, 0, RIFLEMAN_COST, lambda: self.spawn_unit("rifleman", world_center)): return True
                if check(60, 0, ROCKET_COST, lambda: self.spawn_unit("rocket", world_center)): return True
                if check(0, -60, 0, lambda: setattr(self, 'base_menu_mode', 'main')): return True
        elif self.selected_entity.b_type == "factory":
            if check(0, -80, ARMORED_COST, lambda: self.spawn_unit("armored", world_center)): return True
            # (6) M1A2 按钮 (右侧)
            if check(80, -20, M1A2_COST, lambda: self.spawn_unit("m1a2", world_center)): return True
        
        elif self.selected_entity.b_type == "airfield":
            if check(0, -80, HELICOPTER_COST, lambda: self.spawn_unit("helicopter", world_center)): return True
            if check(60, -20, WZ10_COST, lambda: self.spawn_unit("wz10", world_center)): return True
            if not getattr(self.selected_entity, 'is_upgraded', False):
                if check(-60, 60, UPGRADE_AIRFIELD_COST, lambda: setattr(self.selected_entity, 'is_upgraded', True)): return True
            else:
                if check(-60, 60, KIROV_COST, lambda: self.spawn_unit("kirov", world_center)): return True
        elif self.selected_entity.b_type == "radar":
            if check_no_deduct(0, -80, AC130_COST, lambda: setattr(self, 'call_airstrike_mode', True)): return True
            if check_no_deduct(0, 80, MISSILE_COST, lambda: setattr(self, 'call_missile_mode', True)): return True
        elif self.selected_entity.b_type == "research":
            if check(0, -80, MISSILE_DEFENSE_COST, lambda: setattr(self, 'place_building_type', 'missile_def')): return True

            if check(-80, -20, LASER_TURRET_COST, lambda: setattr(self, 'place_building_type', 'laser_turret')): return True

            if check(60, 60, UPGRADE_AIR_COST, lambda: self.upgrade_tech("air")): return True
        return False

    def handle_input(self):
        if self.state == "MENU":
            for event in pygame.event.get():
                if event.type == pygame.QUIT: self.running = False
                if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
                    m_pos = pygame.mouse.get_pos()
                    if self.btn_easy.collidepoint(m_pos): self.difficulty = "EASY"; self.reset_game(); self.state = "PLAYING"
                    elif self.btn_normal.collidepoint(m_pos): self.difficulty = "NORMAL"; self.reset_game(); self.state = "PLAYING"
                    elif self.btn_hard.collidepoint(m_pos): self.difficulty = "HARD"; self.reset_game(); self.state = "PLAYING"
                    elif self.btn_bombard.collidepoint(m_pos): self.bombardment_mode = not self.bombardment_mode
                    elif self.btn_tutorial.collidepoint(m_pos): self.state = "TUTORIAL"
                    elif self.btn_quit.collidepoint(m_pos): self.running = False
            return

        if self.state == "TUTORIAL":
            for event in pygame.event.get():
                if event.type == pygame.QUIT: self.running = False
                if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1:
                    m_pos = pygame.mouse.get_pos()
                    if self.btn_close_tut.collidepoint(m_pos): self.state = "MENU"
            return

        if self.state == "GAMEOVER":
            for event in pygame.event.get():
                if event.type == pygame.QUIT: self.running = False

class Unit(Entity):
    def __init__(self, game, pos, team):
        super().__init__(game, pos, team)
        self.speed = 0
        self.target_pos = None
        self.target_entity = None
        self.attack_cooldown = 0
        self.attack_range = 0
        self.angle = 0
        self.ai_mode = "GUARD" 
         # --- (新) 应用科技升级 ---
        # 只有玩家享受科技加成 (或者可以给AI也加，这里简化只给玩家)
        self.tech_bonus_dmg = 1.0
        if team == "player":
            # 判断单位类型
            tech_type = None
            if isinstance(self, (Rifleman, RocketTrooper)): tech_type = "infantry"
            elif isinstance(self, (ArmoredCar, Harvester)): tech_type = "armor"
            elif isinstance(self, (Helicopter, WZ10, AC130)): tech_type = "air"
            
            if tech_type:
                level = game.tech_levels[tech_type]
                # 每级提升 20% 血量和伤害
                bonus = 1 + (level * 0.2)
                self.max_hp = int(self.max_hp * bonus)
                self.hp = self.max_hp
                self.tech_bonus_dmg = bonus # 攻击时乘这个系数
    def set_target(self, entity):
        if entity.is_air and not self.can_attack_air: return 
        self.target_entity = entity
        self.target_pos = None

    def move_to(self, pos):
        self.target_pos = vec(pos)
        self.target_entity = None

    def find_nearest_enemy(self):
        nearest = None
        min_dist = 99999
        target_team = "player" if self.team == "enemy" else "enemy"
        
        valid_targets = []
        for u in self.game.all_units:
            if u.team == target_team and u.alive:
                if u.is_air and not self.can_attack_air: continue
                valid_targets.append(u)
        
        if self.team == "enemy" and self.ai_mode == "ATTACK" and not valid_targets:
            if self.game.player_base and self.game.player_base.alive: return self.game.player_base

        if not valid_targets:
            for b in self.game.all_buildings:
                if b.team == target_team and b.alive: valid_targets.append(b)

        for e in valid_targets:
            dist = self.pos.distance_to(e.pos)
            if dist < min_dist: min_dist = dist; nearest = e
        return nearest

    def update(self):
        # --- (1) 修复朝向 Bug ---
        # 我们需要一个变量来记录"本帧试图面向的方向"
        face_vec = None
        
        # 1. 攻击逻辑
        if self.target_entity:
            if not self.target_entity.alive: self.target_entity = None
            else:
                dist = self.pos.distance_to(self.target_entity.pos)
                if dist <= self.attack_range: 
                    self.perform_attack(self.target_entity)
                    face_vec = self.target_entity.pos - self.pos # 攻击时面向敌人
                else:
                    move_vec = self.target_entity.pos - self.pos
                    self.pos += move_vec.normalize() * self.speed
                    face_vec = move_vec # 追击时面向敌人
        
        elif self.target_pos:
            move_vec = self.target_pos - self.pos
            enemy = self.find_nearest_enemy()
            if enemy and self.pos.distance_to(enemy.pos) <= self.attack_range:
                self.perform_attack(enemy)
                face_vec = enemy.pos - self.pos # 移动中停下来打人，面向敌人
            else:
                if move_vec.length() < 5: self.target_pos = None
                else: 
                    self.pos += move_vec.normalize() * self.speed
                    face_vec = move_vec # 纯移动，面向目的地
        
        else:
            if not isinstance(self, Harvester) and not isinstance(self, CruiseMissile):
                enemy = self.find_nearest_enemy()
                if enemy:
                    dist = self.pos.distance_to(enemy.pos)
                    if dist <= self.attack_range: 
                        self.perform_attack(enemy)
                        face_vec = enemy.pos - self.pos # 待机攻击，面向敌人
                    else:
                        # 简单的索敌接近
                        move_vec = enemy.pos - self.pos
                        self.pos += move_vec.normalize() * self.speed
                        face_vec = move_vec
        
        # 应用朝向
        if face_vec and face_vec.length() > 0:
            self.angle = math.degrees(math.atan2(-face_vec.y, face_vec.x))

        self.update_rect()
        if self.attack_cooldown > 0: self.attack_cooldown -= 1

        if self.attack_cooldown <= 0:
            self.fire_projectile(target)

class CruiseMissile(Unit):
    def __init__(self, game, target_pos, team):
        spawn_pos = vec(target_pos) + vec(-1000, -1000)
        super().__init__(game, spawn_pos, team)
        self.target_dest = vec(target_pos)
        self.speed = 18 
        self.hp = 100
        self.max_hp = 100
        self.is_mechanical = True
        self.is_air = True
        self.radius = 10
        self.rect = pygame.Rect(self.pos.x, self.pos.y, 1, 1)
        self.angle = 0

    def update(self):
        vec_to_target = self.target_dest - self.pos
        dist = vec_to_target.length()
        if dist < 10:
            self.explode()
            return
        if dist < 300: self.speed = max(4, self.speed * 0.95) 
        direction = vec_to_target.normalize()
        self.pos += direction * self.speed
        self.angle = math.degrees(math.atan2(-direction.y, direction.x))
        self.update_rect()
        self.game.particles.add(Particle(self.game.particles, self.pos, ORANGE, 1, 20, 4))
        self.game.particles.add(Particle(self.game.particles, self.pos, (100, 100, 100), 1, 40, 6))

    def explode(self):
        self.alive = False
        for _ in range(30): Particle(self.game.particles, self.pos, ORANGE, random.uniform(2, 6), 50, 8)
        for _ in range(20): Particle(self.game.particles, self.pos, RED, random.uniform(1, 4), 60, 10)
        for _ in range(10): Particle(self.game.particles, self.pos, WHITE, random.uniform(0.5, 2), 40, 15)
        self.game.particles.add(FloatingText(self.pos, "BOOM!!", RED, 90))
        targets = self.game.all_units + self.game.all_buildings
        for t in targets:
            if t.alive:
                # 如果导弹是中立的，炸所有人；否则只炸敌人
                if self.team == "neutral" or t.team != self.team:
                    if self.pos.distance_to(t.pos) <= 120:
                        t.take_damage(DMG_MISSILE, "heavy")
    def draw(self, surface, offset, zoom):
        screen_pos = to_screen(self.pos, offset, zoom)
        temp_s = pygame.Surface((40, 20), pygame.SRCALPHA)
        pygame.draw.rect(temp_s, WHITE, (5, 5, 30, 10))
        pygame.draw.polygon(temp_s, RED, [(35, 5), (40, 10), (35, 15)])
        pygame.draw.polygon(temp_s, (100, 100, 100), [(5, 5), (0, 0), (5, 10)])
        pygame.draw.polygon(temp_s, (100, 100, 100), [(5, 15), (0, 20), (5, 10)])
        scaled_img = scale_img(temp_s, zoom)
        rotated_img = pygame.transform.rotate(scaled_img, self.angle)
        new_rect = rotated_img.get_rect(center=screen_pos)
        surface.blit(rotated_img, new_rect.topleft)

class Harvester(Unit):
    def __init__(self, game, pos, team, home_building):
        super().__init__(game, pos, team)
        self.home_building = home_building
        self.speed = HARVESTER_STATS[1]
        self.hp = HARVESTER_STATS[2]
        self.max_hp = self.hp
        self.is_mechanical = True
        self.radius = 15
        self.rect = pygame.Rect(pos.x - 15, pos.y - 15, 30, 30)
        self.state = "IDLE" 
        self.mine_target = None
        self.mine_timer = 0
        self.mine_duration = 300 

    def update(self):
        if not self.home_building.alive: self.hp = 0 
        if self.state == "IDLE":
            offset = vec(random.randint(100, 300), 0).rotate(random.randint(0, 360))
            self.mine_target = self.home_building.pos + offset
            self.mine_target.x = max(50, min(self.mine_target.x, WORLD_WIDTH-50))
            self.mine_target.y = max(50, min(self.mine_target.y, WORLD_HEIGHT-50))
            self.state = "MOVING_TO_ORE"
        elif self.state == "MOVING_TO_ORE":
            dist = self.pos.distance_to(self.mine_target)
            if dist < 5:
                self.state = "MINING"
                self.mine_timer = self.mine_duration
            else: self.target_pos = self.mine_target
        elif self.state == "MINING":
            self.mine_timer -= 1
            if self.mine_timer <= 0: self.state = "RETURNING"
        elif self.state == "RETURNING":
            dist = self.pos.distance_to(self.home_building.pos)
            if dist < 40: 
                # --- 修复：增加上限限制 ---
                if self.team == "player":
                    self.game.cmd_exp = min(self.game.cmd_exp + HARVEST_AMOUNT, self.game.max_cmd_exp)
                else:
                    # 敌军通常资源无限或者有自己的逻辑，但也可以加个软上限防止溢出
                    self.game.enemy_exp += HARVEST_AMOUNT
                
                self.state = "IDLE" 
            else: self.target_pos = self.home_building.pos
        super().update()
    
    def draw(self, surface, offset, zoom):
        screen_pos = to_screen(self.pos, offset, zoom)
        w = scale(24, zoom)
        h = scale(16, zoom)
        color = YELLOW
        if self.state == "RETURNING": color = (255, 223, 0) 
        pygame.draw.rect(surface, color, (screen_pos[0]-w//2, screen_pos[1]-h//2, w, h))
        pygame.draw.rect(surface, BLACK, (screen_pos[0]-w//2, screen_pos[1]-h//2, w, h), 1)
        if self.state == "MINING":
            t = (pygame.time.get_ticks() // 200) % 2
            if t == 0: pygame.draw.circle(surface, WHITE, screen_pos, scale(12, zoom), 1)
        self.draw_health_and_marker(surface, offset, zoom)

class Engineer(Unit):

        self.speed = ENGINEER_SPEED
        self.build_pos = vec(build_pos)
        self.is_building = False
        self.build_timer = 0
        self.hp = 30
        self.max_hp = 30 
        self.radius = 10
        self.rect = pygame.Rect(pos.x - 10, pos.y - 10, 20, 20)

    def update(self):
        if not self.is_building:
            # ... (移动逻辑不变) ...
            dist = self.pos.distance_to(self.build_pos)
            if dist < 5: self.is_building = True
            else: self.pos += (self.build_pos - self.pos).normalize() * self.speed
        else:
            self.build_timer += 1
            if self.build_timer >= BUILD_TIME * FPS:
                new_b = None
                if hasattr(self, 'target_type'):
                    if self.target_type == 'radar': new_b = RadarBuilding(self.game, self.build_pos, self.team)

            # 坠毁过程中的小爆炸
            if random.randint(0, 10) == 0:
                offset = vec(random.randint(-60, 60), random.randint(-20, 20))
                self.game.particles.add(Particle(self.game.particles, self.pos + offset, ORANGE, 2, 30, 8))

            return

            self.deploy_timer -= 1
            if self.deploy_timer > 0:
                return 
            else:
                self.deployed = True
                self.game.particles.add(FloatingText(self.pos, "AIRSHIP READY", GREEN_HAT))

            self.crash_timer = 180 # 3秒坠毁动画 (更长)

        self.capture_progress = 0
        self.capturing_team = None
        self.max_capture = OIL_CAPTURE_TIME

    def update(self):
        if self.team != "neutral":
            self.spawn_timer += 1
            if self.spawn_timer > MERC_FACTORY_INTERVAL:
                self.spawn_timer = 0
                spawn_pos = self.pos + vec(0, 70)
                # 生成装甲车
                self.game.all_units.append(ArmoredCar(self.game, spawn_pos, self.team))
                self.game.particles.add(FloatingText(self.pos, "+TANK", YELLOW))

        # 占领逻辑
        capturer = None
        min_dist = 100
        for u in self.game.all_units:
            if u.alive and not u.is_air:
                dist = self.pos.distance_to(u.pos)
                if dist < min_dist:
                    capturer = u
                    break
        if capturer:
            if capturer.team != self.team:
                if self.capturing_team != capturer.team:
                    self.capturing_team = capturer.team
                    self.capture_progress = 0 
                self.capture_progress += 1
                if self.capture_progress >= self.max_capture:
                    self.team = capturer.team
                    self.game.particles.add(FloatingText(self.pos, "FACTORY CAPTURED!", BLUE if self.team=="player" else RED))
            else: self.capture_progress = 0

        screen_pos = to_screen(self.pos, offset, zoom)
        w = scale(80, zoom)

        if self.capture_progress > 0 and self.capturing_team:
            bar_w = scale(40, zoom)

            pct = self.capture_progress / self.max_capture

class AntiAirTurret(Building):
    def __init__(self, game, pos, team):
        super().__init__(game, pos, team, "aa_turret")
        self.hp = AA_TURRET_STATS[2]
        self.max_hp = self.hp
        self.radius = 25
        self.rect = pygame.Rect(pos.x - 25, pos.y - 25, 50, 50)
        self.attack_range = AA_TURRET_STATS[0]
        self.attack_cooldown = 0
        self.angle = 0
        self.target = None

    def update(self):
        if not self.target or not self.target.alive or not self.target.is_air:
            self.target = self.find_air_target()
        
        if self.target:
            dist = self.pos.distance_to(self.target.pos)
            if dist <= self.attack_range:
                vec_to = self.target.pos - self.pos
                self.angle = math.degrees(math.atan2(-vec_to.y, vec_to.x))
                if self.attack_cooldown <= 0:
                    self.fire()
                    self.attack_cooldown = 20 
            else:
                self.target = None 
        if self.attack_cooldown > 0: self.attack_cooldown -= 1

    def find_air_target(self):
        nearest = None
        min_dist = 99999
        enemies = [u for u in self.game.all_units if u.team != self.team and u.is_air]
        for e in enemies:
            dist = self.pos.distance_to(e.pos)
            if dist < min_dist: min_dist = dist; nearest = e
        return nearest

    def fire(self):
        proj = Projectile(self.game, self.pos, self.target, self.team, DMG_AA, "heavy")
        proj.speed = 15 
        self.game.projectiles.append(proj)

    def draw(self, surface, offset, zoom):

def to_screen(pos, offset, zoom):
    return (int((pos[0] - offset.x) * zoom), int((pos[1] - offset.y) * zoom))

def scale(value, zoom):

class FloatingText(pygame.sprite.Sprite):
    def __init__(self, pos, text, color, duration=60):
        super().__init__()

        self.text = text
        self.color = color
        self.timer = duration
        self.vel = vec(0, -0.5) 
        self.font = pygame.font.SysFont("Arial", 20, bold=True)

    def update(self):
        self.pos += self.vel
        self.timer -= 1
        if self.timer <= 0: self.kill()

    def draw(self, surface, offset, zoom):
        screen_pos = to_screen(self.pos, offset, zoom)
        surf = self.font.render(self.text, True, self.color)
        outline = self.font.render(self.text, True, BLACK)
        rect = surf.get_rect(center=screen_pos)
        surface.blit(outline, (rect.x+1, rect.y+1))
        surface.blit(surf, rect)

class Particle(pygame.sprite.Sprite):
    def __init__(self, groups, pos, color, speed, life, size, decay=True):
        super().__init__(groups)
        self.pos = vec(pos)
        self.vel = vec(random.uniform(-1, 1), random.uniform(-1, 1)).normalize() * speed
        self.color = color
        self.life = life
        self.max_life = life
        self.size = size
        self.decay = decay
    def update(self):
        self.pos += self.vel
        self.life -= 1
        if self.life <= 0: self.kill()
    def draw(self, surface, offset, zoom):
        screen_pos = to_screen(self.pos, offset, zoom)
        if not (-50 <= screen_pos[0] <= SCREEN_WIDTH+50 and -50 <= screen_pos[1] <= SCREEN_HEIGHT+50): return
        if self.decay: current_size = max(0, int(self.size * (self.life / self.max_life)))
        else: current_size = self.size
        scaled_size = scale(current_size, zoom)
        if scaled_size > 0:
            pygame.draw.circle(surface, self.color, screen_pos, scaled_size)

        self.hp = 100
        self.max_hp = 100

        self.is_mechanical = False
        self.is_air = False
        self.can_attack_air = False

    def update_rect(self):
        self.rect.center = (int(self.pos.x), int(self.pos.y))

    def take_damage(self, amount, damage_type="normal"):
        self.hp -= amount
        if self.hp <= 0: self.die()

    def die(self):
        self.alive = False
        self.create_death_effect()

    def create_death_effect(self):
        count = 15 if self.is_mechanical else 10
        color = ORANGE if self.is_mechanical else RED
        speed = 3 if self.is_mechanical else 1.5
        size = 5 if self.is_mechanical else 3
        for _ in range(count):
            Particle(self.game.particles, self.pos, color, random.uniform(0.5, speed), random.randint(20, 40), size)
        if self.is_mechanical:
            for _ in range(5): Particle(self.game.particles, self.pos, (50, 50, 50), 1, 60, 6)

    def draw_health_and_marker(self, surface, offset, zoom, y_offset=-25):
        screen_pos = to_screen(self.pos, offset, zoom)
        bar_w = scale(30, zoom)
        bar_h = scale(4, zoom)
        scaled_y_offset = scale(y_offset, zoom)
        if self.hp < self.max_hp:
            ratio = max(0, self.hp / self.max_hp)
            pygame.draw.rect(surface, RED, (screen_pos[0] - bar_w//2, screen_pos[1] + scaled_y_offset, bar_w, bar_h))
            pygame.draw.rect(surface, (0, 255, 0), (screen_pos[0] - bar_w//2, screen_pos[1] + scaled_y_offset, bar_w * ratio, bar_h))

        self.is_aoe = is_aoe
        self.aoe_radius = aoe_radius
        self.target_dest = None
        
        self.speed = ROCKET_SPEED if d_type == "rocket" else BULLET_SPEED
        if d_type == "ac130_vis": self.speed = 18 
        # (6) 坦克炮弹速度
        if d_type == "tank_shell": self.speed = 10 
        
        if d_type == "laser":
            self.life_time = 5 
            self.target_dest = target.pos if target else pos 
        else:
            self.life_time = 80
            if target: self.vel = (target.pos - pos).normalize() * self.speed
            else: self.vel = vec(0,0)

    def update(self):
        if self.d_type == "laser":
            self.life_time -= 1
            if self.life_time <= 0: self.alive = False
            return 
        self.pos += self.vel
        self.life_time -= 1
        if self.d_type == "ac130_vis":
            if self.target_dest and self.pos.distance_to(self.target_dest) <= self.speed:
                self.alive = False 
            elif self.life_time <= 0:
                self.alive = False
            return 
        if self.target_dest:
            if self.pos.distance_to(self.target_dest) <= self.speed:
                self.on_hit(None); return
        if self.life_time <= 0: self.alive = False; return
        hit_list = self.game.all_units + self.game.all_buildings
        for entity in hit_list:
            if entity.team != self.team and entity.alive and entity.team != "neutral":
                if entity.rect.collidepoint(self.pos): self.on_hit(entity); return
